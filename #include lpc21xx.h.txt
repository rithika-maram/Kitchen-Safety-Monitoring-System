#include <lpc21xx.h>
#include "types.h"
#include "delay.h"
#include "LCD.h"
#include "KPM.h"

#define EINT0_LED       3
#define EINT0_SWITCH    16
#define EINT0_CHNO      14

#define PASSWORD        1234

volatile u32 edit_request = 0;


/* EINT0 interrupt service routine */
void eint0_isr(void) __irq
{
    EXTINT = 1 << 0;       // Clear EINT0 interrupt flag
    edit_request = 1;      // Notify main program
    VICVectAddr = 0;       // Acknowledge VIC
}


/* Configure EINT0 on P0.16 */
void eint0_enable(void)
{
    /* Select EINT0 function for P0.16 */
    PINSEL1 &= ~(3 << 0);
    PINSEL1 |=  (1 << 0);

    /* Falling-edge triggered interrupt */
    EXTMODE |= 1 << 0;
    EXTPOLAR &= ~(1 << 0);

    /* Clear any pending EINT0 interrupt */
    EXTINT = 1 << 0;

    /* Configure EINT0 as an IRQ */
    VICIntSelect &= ~(1 << EINT0_CHNO);

    /* Configure vector slot 0 */
    VICVectAddr0 = (u32)eint0_isr;
    VICVectCntl0 = (1 << 5) | EINT0_CHNO;

    /* Enable EINT0 */
    VICIntEnable |= 1 << EINT0_CHNO;
}


/* Read four keypad digits and return the entered number */
u32 ReadPassword(void)
{
    u32 entered = 0;
    u32 count = 0;
    u8 key;

    WRITE_LCD_CMD(0x80);
    StrLCD((u8 *)"Enter 4-digit PIN");

    WRITE_LCD_CMD(0xC0);
    StrLCD((u8 *)"PIN: ");

    while (count < 4)
    {
        key = KeyScan();

        if (key >= '0' && key <= '9')
        {
            entered = (entered * 10) + (key - '0');
            WRITE_LCD_DATA('*');
            count++;
        }
        else if (key == 'C')
        {
            entered = 0;
            count = 0;

            WRITE_LCD_CMD(0xC0);
            StrLCD((u8 *)"                ");
            WRITE_LCD_CMD(0xC0);
            StrLCD((u8 *)"PIN: ");
        }
    }

    return entered;
}


/* Password verification */
void PasswordCheck(void)
{
    u32 entered;
    u32 attempts = 0;

    while (attempts < 3)
    {
        entered = ReadPassword();

        if (entered == PASSWORD)
        {
            WRITE_LCD_CMD(0x80);
            StrLCD((u8 *)"Access Granted");

            delay_ms(1000);

            WRITE_LCD_CMD(0x80);
            StrLCD((u8 *)"Edit Menu");

            /* Add your edit menu handling here */
            return;
        }
        else
        {
            attempts++;

            WRITE_LCD_CMD(0x80);
            StrLCD((u8 *)"Wrong Password");

            WRITE_LCD_CMD(0xC0);

            if (attempts < 3)
            {
                StrLCD((u8 *)"Try Again");
            }
            else
            {
                StrLCD((u8 *)"Access Locked");
            }

            delay_ms(1500);
        }
    }

    /* Lock the system after 3 wrong attempts */
    while (1);
}


int main(void)
{
    Init_LCD();
    InitKPM();

    /* Optional LED output */
    IODIR0 |= 1 << EINT0_LED;

    WRITE_LCD_CMD(0x80);
    StrLCD((u8 *)"System Ready");

    WRITE_LCD_CMD(0xC0);
    StrLCD((u8 *)"Press Switch1");

    eint0_enable();

    while (1)
    {
        if (edit_request == 1)
        {
            edit_request = 0;
            PasswordCheck();

            /* Display standby message after returning */
            WRITE_LCD_CMD(0x80);
            StrLCD((u8 *)"System Ready");

            WRITE_LCD_CMD(0xC0);
            StrLCD((u8 *)"Press Switch1");
        }
    }
}
